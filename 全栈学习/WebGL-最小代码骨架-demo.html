const canvas = document.querySelector('#gl');
const gl = canvas.getContext('webgl');

function compile(type, source) {
  const shader = gl.createShader(type);
  gl.shaderSource(shader, source);
  gl.compileShader(shader);
  if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
    throw new Error(gl.getShaderInfoLog(shader));
  }
  return shader;
}

function createProgram(vsSource, fsSource) {
  const program = gl.createProgram();
  gl.attachShader(program, compile(gl.VERTEX_SHADER, vsSource));
  gl.attachShader(program, compile(gl.FRAGMENT_SHADER, fsSource));
  gl.linkProgram(program);
  return program;
}

// buffer：3 个二维顶点
const vertices = new Float32Array([
  -0.65, -0.55,
   0.65, -0.55,
   0.00,  0.72,
]);

const buffer = gl.createBuffer();
gl.bindBuffer(gl.ARRAY_BUFFER, buffer);
gl.bufferData(gl.ARRAY_BUFFER, vertices, gl.STATIC_DRAW);

// 每帧：清屏 → 使用 program → 设置 uniform → 绑定 attribute → 绘制
gl.clearColor(0.05, 0.08, 0.11, 1);
gl.clear(gl.COLOR_BUFFER_BIT);
gl.useProgram(program);
gl.uniformMatrix4fv(uProjection, false, projection);
gl.uniformMatrix4fv(uView, false, view);
gl.uniformMatrix4fv(uModel, false, model);
gl.bindBuffer(gl.ARRAY_BUFFER, buffer);
gl.enableVertexAttribArray(aPosition);
gl.vertexAttribPointer(aPosition, 2, gl.FLOAT, false, 0, 0);
gl.drawArrays(gl.TRIANGLES, 0, 3);
展开：每一步对应什么 WebGL API
步骤 关键 API 作用
获取上下文 canvas.getContext('webgl') 拿到操作 GPU 的入口对象。
编译 shader createShader / shaderSource / compileShader 把 GLSL 字符串编译成 GPU 可执行的着色器。
链接 program createProgram / attachShader / linkProgram 把顶点 shader 和片元 shader 组成渲染程序。
上传顶点 createBuffer / bindBuffer / bufferData 把 CPU 里的顶点数组复制到 GPU buffer。
取 location getAttribLocation / getUniformLocation 拿到 shader 中变量的位置引用。
传 uniform uniformMatrix4fv / uniform1f / uniform1i 每帧传矩阵、时间、模式等参数。
绑定 attribute enableVertexAttribArray / vertexAttribPointer 告诉 WebGL 顶点数据如何读取。
绘制 drawArrays 真正提交绘制命令。
